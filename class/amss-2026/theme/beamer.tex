% Characters the text font lacks (such as → in Noto Sans, or ≥ in Latin
% Modern Sans) come from DejaVu, if installed, and from Latin Modern Math.
\ifLuaTeX
  \IfFontExistsTF{DejaVu Sans}%
    {\directlua{luaotfload.add_fallback("amssfallback", {"DejaVu Sans:mode=node;", "Latin Modern Math:mode=node;"})}}%
    {\directlua{luaotfload.add_fallback("amssfallback", {"Latin Modern Math:mode=node;"})}}
  \IfFontExistsTF{DejaVu Sans Mono}%
    {\directlua{luaotfload.add_fallback("amssmonofallback", {"DejaVu Sans Mono:mode=node;", "Latin Modern Math:mode=node;"})}}%
    {\directlua{luaotfload.add_fallback("amssmonofallback", {"Latin Modern Math:mode=node;"})}}
  \IfFontExistsTF{Noto Sans}%
    {\setsansfont{Noto Sans}[Scale=0.94,RawFeature={fallback=amssfallback}]}%
    {\setsansfont{Latin Modern Sans}[RawFeature={fallback=amssfallback}]}
  \IfFontExistsTF{Noto Sans Mono}%
    {\setmonofont{Noto Sans Mono}[Scale=0.9,RawFeature={fallback=amssmonofallback}]}%
    {\setmonofont{Latin Modern Mono}[RawFeature={fallback=amssmonofallback}]}
\fi

% Frame title over a hairline that starts with a short teal segment. The
% struts give every title line the same height and depth, so the rule sits at
% the same place on every frame.
\newcommand{\amsstitlestrut}{\rule[-0.3\baselineskip]{0pt}{1.08\baselineskip}}
\setbeamertemplate{frametitle}{%
  \nointerlineskip\vskip 2.9mm
  \begin{beamercolorbox}[wd=\paperwidth]{frametitle}%
    \leftskip=\amssmargin \rightskip=\amssmargin plus 1fil\relax
    \usebeamerfont{frametitle}\amsstitlestrut\insertframetitle\amsstitlestrut\par
    \nointerlineskip\vskip 0.4mm
    \moveright\amssmargin\hbox{%
      \textcolor{amssteal}{\rule{12mm}{1.1pt}}%
      \textcolor{amssline}{\rule{\dimexpr\paperwidth-2\amssmargin-12mm\relax}{0.4pt}}}%
  \end{beamercolorbox}%
  \vskip 1mm}

% Course label and frame number; the cover has neither.
\newif\ifamsscover
\setbeamertemplate{footline}{%
  \ifamsscover\else
    \leavevmode\hbox to\paperwidth{%
      \hskip\amssmargin
      \usebeamerfont{footline}\usebeamercolor[fg]{footline}%
      \insertshortinstitute\hfill\insertframenumber\,/\,\inserttotalframenumber
      \hskip\amssmargin}%
    \vskip 3.5mm
  \fi}

% Cover: logo on a white plate, then session, title, course and author, on
% navy. The flag is cleared once the cover page has been shipped out.
\setbeamertemplate{background canvas}{%
  \ifamsscover\color{amssnavy}\vrule width\paperwidth height\paperheight\fi}
\makeatletter
\setbeamertemplate{title page}{%
  \global\amsscovertrue
  \AddToHookNext{shipout/after}{\global\amsscoverfalse}%
  % Beamer keeps room for the footline, which the cover does not have.
  \vskip\footheight plus .5fill\relax
  \ifx\inserttitlegraphic\@empty\else
    \begin{minipage}[c]{40mm}%
      \tikz\node[fill=white,rounded corners=2.2mm,inner sep=2.4mm]{\resizebox{34mm}{!}{\inserttitlegraphic}};%
    \end{minipage}\hfill
  \fi
  \begin{minipage}[c]{\dimexpr\textwidth-46mm\relax}\raggedright
    \ifx\insertsubtitle\@empty\else
      {\usebeamerfont{subtitle}\color{amsstealight}\insertsubtitle\par}\vskip 1.5mm
    \fi
    {\usebeamerfont{title}\color{white}%
     % Lines of similar length, as on the HTML cover: no single word on the last
     % one. The word spaces are rigid, so that a title a little wider than the
     % line is broken instead of being squeezed onto it.
     \rightskip=0pt plus 0.7\linewidth \parfillskip=0pt
     \spaceskip=\fontdimen2\font \xspaceskip=\spaceskip
     \hyphenpenalty=10000 \exhyphenpenalty=10000\relax
     \inserttitle\par}%
    \vskip 6mm
    \ifx\insertinstitute\@empty\else
      {\usebeamerfont{institute}\color{amsstealight}\insertinstitute\par}\vskip 1mm
    \fi
    {\usebeamerfont{author}\color{white}\insertauthor\par}%
  \end{minipage}}
\makeatother

% Bullets echo the rounded nodes of the course logo.

\setbeamertemplate{itemize subitem}{\tikz[baseline=-0.1ex]\draw[amssteal,line width=0.12ex,rounded corners=0.1ex] (0,0) rectangle (0.52ex,0.52ex);}
\setbeamertemplate{itemize subsubitem}{\textcolor{amssteal}{--}}

% Quotations: upright, ragged-right, unhyphenated text beside a teal bar.
% theme/amss.lua wraps the epigraph in amssepigraph (larger, with balanced
% lines) and its source line in amsssource (small and muted).
\newcommand{\amssquotesetup}{}
\newenvironment{amssepigraph}{%
  \setbeamercolor{quote}{fg=amssnavy}%
  \setbeamerfont{quote}{shape=\upshape,series=\bfseries,size=\Large}%
  \renewcommand{\amssquotesetup}{\rightskip=0pt plus 0.45\linewidth \parfillskip=0pt plus 0.45\linewidth\relax}}%
 {\par\vskip 1.5mm}
\newenvironment{amsssource}{\begingroup\footnotesize\color{amssmuted}}{\par\endgroup}
\newsavebox{\amssquotebox}
\renewenvironment{quote}{%
  \par\begin{lrbox}{\amssquotebox}%
  \begin{minipage}{\dimexpr\linewidth-4.6mm\relax}%
  \raggedright\hyphenpenalty=10000 \exhyphenpenalty=10000
  \usebeamerfont{quote}\usebeamercolor[fg]{quote}\amssquotesetup\strut\ignorespaces}%
 {\ifhmode\unskip\strut\fi\end{minipage}\end{lrbox}%   after a list or code: no empty last line
  \noindent{\color{amssteal}\vrule width 1.4pt}\hskip 3.2mm\usebox{\amssquotebox}\par\vskip 0.3ex}
