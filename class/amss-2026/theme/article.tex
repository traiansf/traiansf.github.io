% AMSS course look for the printable guides (Lab 0; see lab/Makefile). The
% palette is shared with static/amss.css. Noto Sans is used when it is
% installed; otherwise the text falls back to Latin Modern Sans.
\usepackage{framed}
\usepackage{fancyhdr}

\definecolor{amssnavy}{HTML}{0A2145}
\definecolor{amssteal}{HTML}{087373}
\definecolor{amssmuted}{HTML}{4A5A60}
\definecolor{amssline}{HTML}{D5E0E3}
\definecolor{shadecolor}{HTML}{F3F7F8}

% Characters the text font lacks (such as → in Noto Sans) come from DejaVu
% Sans, if installed, and from Latin Modern Math.
\ifLuaTeX
  \IfFontExistsTF{DejaVu Sans}%
    {\directlua{luaotfload.add_fallback("amssfallback", {"DejaVu Sans:mode=node;", "Latin Modern Math:mode=node;"})}}%
    {\directlua{luaotfload.add_fallback("amssfallback", {"Latin Modern Math:mode=node;"})}}
  \IfFontExistsTF{DejaVu Sans Mono}%
    {\directlua{luaotfload.add_fallback("amssmonofallback", {"DejaVu Sans Mono:mode=node;", "Latin Modern Math:mode=node;"})}}%
    {\directlua{luaotfload.add_fallback("amssmonofallback", {"Latin Modern Math:mode=node;"})}}
  \IfFontExistsTF{Noto Sans}{%
    \setmainfont{Noto Sans}[Scale=0.93,RawFeature={fallback=amssfallback}]%
    \setsansfont{Noto Sans}[Scale=0.93,RawFeature={fallback=amssfallback}]}{%
    \renewcommand{\familydefault}{\sfdefault}}
  \IfFontExistsTF{Noto Sans Mono}{\setmonofont{Noto Sans Mono}[Scale=0.84,RawFeature={fallback=amssmonofallback}]}{}
\fi
\linespread{1.08}

\makeatletter
% The course line arrives as the subtitle (theme/amss.lua). Pandoc's own
% \subtitle is a \providecommand, so this definition is the one used.
\newcommand{\amsscourse}{}
\newcommand{\subtitle}[1]{\renewcommand{\amsscourse}{#1}}
\renewcommand{\maketitle}{%
  {\raggedright
   \ifx\amsscourse\@empty\else
     {\small\bfseries\color{amssteal}\amsscourse\par}\vskip 2.5mm
   \fi
   {\LARGE\bfseries\color{amssnavy}\@title\par}%
   \vskip 2.5mm
   {\color{amssmuted}\@author\par}%
   \vskip 4mm
   {\color{amssline}\hrule height 0.8pt}%
   \vskip 5mm}}

\renewcommand{\section}{\@startsection{section}{1}{\z@}%
  {-4.2ex \@plus -1ex \@minus -.2ex}{1.5ex \@plus .2ex}%
  {\normalfont\Large\bfseries\color{amssnavy}\raggedright}}
\renewcommand{\subsection}{\@startsection{subsection}{2}{\z@}%
  {-3.2ex \@plus -1ex \@minus -.2ex}{1.1ex \@plus .2ex}%
  {\normalfont\large\bfseries\color{amssnavy}\raggedright}}

\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[L]{\footnotesize\color{amssmuted}\@title}
\fancyfoot[R]{\footnotesize\color{amssmuted}\thepage}

% Code on a tinted panel.

    \renewenvironment{Shaded}{\begin{snugshade}\small}{\end{snugshade}}}}
\makeatother
