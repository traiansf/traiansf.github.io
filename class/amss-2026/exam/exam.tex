  {-1ex}{1.2ex \@plus .2ex}%

% A question and its options stay on the same page.
\newenvironment{amssquestion}{\par\addvspace{1.6ex}\begin{minipage}{\linewidth}\setlength{\parskip}{0pt}}%
  {\end{minipage}\par}
\newenvironment{amssoptions}{\begin{list}{}{%
    \setlength{\leftmargin}{2.4em}\setlength{\labelwidth}{1.6em}%
    \setlength{\labelsep}{0.8em}\setlength{\itemsep}{0.1ex}%
    \setlength{\parsep}{0pt}\setlength{\topsep}{0.5ex}}\raggedright}%
  {\end{list}}
